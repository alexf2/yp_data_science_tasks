class OutlierClipper(BaseEstimator, TransformerMixin): - стейтфул трансформ
    def __init__(self, factor=1.5):
        self.factor = factor
        
    def fit(self, X, y=None):
        # Learn stateful metrics (bounds) from training data
        q25, q75 = np.percentile(X, [25, 75], axis=0)
        iqr = q75 - q25
        self.lower_bound_ = q25 - (self.factor * iqr)
        self.upper_bound_ = q75 + (self.factor * iqr)
        return self
        
    def transform(self, X):
        # Use learned states to transform any dataset identically
        return np.clip(X, self.lower_bound_, self.upper_bound_)
